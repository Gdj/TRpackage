trpFNhasScrollBarY, trpFNhasScrollBarX

스크롤바 생성여부를 컨테이너에 클래스를 출력해주는 함수.

trpFNhasScrollBarY()

설명
사용 : trpFNhasScrollBarY.init('.js-scrollbar-wrap');
class 설명
속성 이름 속성 설명
init($컨테이너) 선택자로 컨테이너를 지정해주면,
그 컨테이너 내부의 .trpHasScrollBarY 요소의 스크롤바 존재 여부에 따라,
this 선택자에 "scrollbar_on" 클래스를 토글한다.
refresh 스크롤바의 존재 여부를 다시 체크하여, this 선택자에 "scrollbar_on" 클래스를 토글한다.
스크롤바의 존재 여부가 바뀌는 상황에서, 이 함수를 호출하여 체크할 수 있다.
class 설명
클래스 이름 클래스 설명
js-TRPscrollbar-wrapper [변경] js-TRPscrollbar-wrapper 스크롤바 컨네이너 선택자로 ".init" 시 변경할수있다.
그 컨테이너 내부의 .trpHasScrollBarY 요소의 스크롤바 존재 여부에 따라,
선택자에 "trpHasScroll_on" 클래스를 토글한다.
trpHasScrollBarY [고정] trpHasScrollBarX [고정] 스크롤바 컨텐츠, 실제 스크롤이 생시는 요소
trpHasScroll_on [고정] 스크롤 여부 클래스,

  <script type="text/javascript">	 
    /* IIFE (scrollbar) */
    const trpFNhasScrollBarY = (() => {
      let elements = [];

      function update(wrapper) {
        const scrollEl = wrapper.querySelector('.trpHasScrollBarY');
        if (!scrollEl) return;

        const check = scrollEl.scrollHeight > scrollEl.clientHeight;
        wrapper.classList.toggle('trpHasScroll_on', check);
      }

      function refresh() {
        elements.forEach(update);
      }

      function init(selector = '.js-TRPscrollbar-wrapper') {
        const newElements = document.querySelectorAll(selector);
        newElements.forEach(wrapper => {
          if (elements.includes(wrapper)) return;
          elements.push(wrapper);
          const scrollEl = wrapper.querySelector('.trpHasScrollBarY');
          if (!scrollEl) return;
          update(wrapper);
          new ResizeObserver(() => {
            update(wrapper);
          }).observe(scrollEl);

        });
      }

      window.addEventListener('resize', refresh);

      return {
        init,
        refresh
      };

    })();

    trpFNhasScrollBarY.init('.js-scrollbar-wrap');
  </script>
